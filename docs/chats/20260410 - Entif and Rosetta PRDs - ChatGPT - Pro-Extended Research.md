export async function authorize(input: {
  principalRef: string;
  capability: string;
  action: string;
  resourceRef: string;
  decision: IamDecision | null;
  policy: PolicyBundle;
}) {
  if (!input.decision) throw new Error("Missing decision token");
  if (input.decision.effect !== "allow") throw new Error("Denied by decision");
  if (new Date(input.decision.expires_at).getTime() < Date.now()) {
    throw new Error("Decision token expired");
  }
  return { ok: true };
}
